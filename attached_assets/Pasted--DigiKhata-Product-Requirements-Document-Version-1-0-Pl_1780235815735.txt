# DigiKhata — Product Requirements Document
**Version:** 1.0 | **Platform:** Anakin.io Hackathon | **Stack:** React + Node/Express

---

## 1. Product Summary

DigiKhata is a proactive personal finance app for first-time earners and college students. It divides salary automatically into fixed budget buckets, tracks real-time spending, and delivers contextual nudges *before* the user overspends — using Anakin.io Wire and Universal Scraper for live price intelligence.

**Core promise:** No guilt. No month-end shock. Just a quiet guide.

---

## 2. Budget Allocation Engine

When a user saves their salary, the system auto-computes allocations:

| Category       | % | Formula             |
|----------------|---|---------------------|
| Rent           | 25 | salary × 0.25      |
| Savings        | 20 | salary × 0.20      |
| Groceries      | 10 | salary × 0.10      |
| Fun & Leisure  | 10 | salary × 0.10      |
| Health         | 5  | salary × 0.05      |
| Ordering In    | 5  | salary × 0.05      |
| Shopping       | 5  | salary × 0.05      |
| Investment     | 20 | salary × 0.20 (remainder) |

**Total: 100%**

```js
// shared/budgetEngine.js
export const ALLOCATIONS = {
  rent:          0.25,
  savings:       0.20,
  groceries:     0.10,
  fun_leisure:   0.10,
  health:        0.05,
  ordering_in:   0.05,
  shopping:      0.05,
  investment:    0.20,
};

export function computeBudget(salary) {
  return Object.fromEntries(
    Object.entries(ALLOCATIONS).map(([k, v]) => [k, Math.round(salary * v)])
  );
}

export function spendingPace(budget, spent, dayOfMonth, daysInMonth) {
  const safePace = (dayOfMonth / daysInMonth) * budget;
  return { safePace, isOverpacing: spent > safePace, remaining: budget - spent };
}
```

### 6a. Profile Page — Salary Input + Auto-Allocation

```jsx
// frontend/src/pages/Profile.jsx
import { useState, useEffect } from "react";
import { computeBudget } from "../../shared/budgetEngine";

const ICONS = {
  rent: "🏠", savings: "🏦", groceries: "🥦",
  fun_leisure: "🎉", health: "💊", ordering_in: "🛵",
  shopping: "🛍️", investment: "📈"
};

export default function Profile() {
  const [salary, setSalary] = useState("");
  const [saved, setSaved] = useState(null);
  const [budget, setBudget] = useState(null);
  const [editing, setEditing] = useState(false);

  useEffect(() => {
    const s = localStorage.getItem("dk_salary");
    if (s) { setSaved(Number(s)); setBudget(computeBudget(Number(s))); }
  }, []);

  function handleSave() {
    const val = Number(salary);
    if (!val || val < 1000) return;
    localStorage.setItem("dk_salary", val);
    fetch("/api/user/salary", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ salary: val })
    });
    setSaved(val);
    setBudget(computeBudget(val));
    setEditing(false);
    setSalary("");
  }

  return (
    <div style={{ background: "var(--bg)", minHeight: "100vh", padding: "24px 16px" }}>
      <h1 style={{ fontSize: 22, fontWeight: 500, marginBottom: 4 }}>Your Profile</h1>
      <p style={{ color: "var(--text-muted)", fontSize: 14, marginBottom: 24 }}>
        Set your salary once. DigiKhata handles the rest.
      </p>

      {/* Salary card */}
      <div style={{
        background: "var(--grad-card)", border: "0.5px solid #E8DDD5",
        borderRadius: "var(--radius-lg)", padding: "20px", marginBottom: 20
      }}>
        {saved && !editing ? (
          <>
            <div style={{ fontSize: 13, color: "var(--text-muted)" }}>Monthly income</div>
            <div style={{ fontSize: 32, fontWeight: 500, margin: "4px 0 12px" }}>
              ₹{saved.toLocaleString("en-IN")}
            </div>
            <button onClick={() => setEditing(true)} style={btnStyle("outline")}>
              Update salary
            </button>
          </>
        ) : (
          <>
            <label style={{ fontSize: 13, color: "var(--text-muted)" }}>
              {saved ? "New monthly income" : "What's your monthly income?"}
            </label>
            <div style={{ display: "flex", gap: 10, marginTop: 8 }}>
              <input
                type="number"
                placeholder="e.g. 35000"
                value={salary}
                onChange={e => setSalary(e.target.value)}
                style={inputStyle}
              />
              <button onClick={handleSave} style={btnStyle("primary")}>Save</button>
              {editing && (
                <button onClick={() => setEditing(false)} style={btnStyle("ghost")}>Cancel</button>
              )}
            </div>
          </>
        )}
      </div>

      {/* Auto-allocation grid */}
      {budget && (
        <>
          <div style={{ fontSize: 13, color: "var(--text-muted)", marginBottom: 12 }}>
            Your money, automatically split:
          </div>
          <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 10 }}>
            {Object.entries(budget).map(([key, amount]) => (
              <AllocationCard key={key} label={key} amount={amount} salary={saved} />
            ))}
          </div>
        </>
      )}
    </div>
  );
}

function AllocationCard({ label, amount, salary }) {
  const pct = Math.round((amount / salary) * 100);
  const nice = label.replace("_", " ");
  return (
    <div style={{
      background: "var(--surface)", border: "0.5px solid #E8DDD5",
      borderRadius: "var(--radius-md)", padding: "14px 16px"
    }}>
      <div style={{ fontSize: 18, marginBottom: 4 }}>{ICONS[label]}</div>
      <div style={{ fontSize: 13, color: "var(--text-muted)", textTransform: "capitalize" }}>{nice}</div>
      <div style={{ fontSize: 18, fontWeight: 500, margin: "2px 0" }}>
        ₹{amount.toLocaleString("en-IN")}
      </div>
      <div style={{ fontSize: 12, color: "var(--text-hint)" }}>{pct}% of income</div>
    </div>
  );
}

const inputStyle = {
  flex: 1, padding: "10px 14px", borderRadius: "var(--radius-sm)",
  border: "0.5px solid #D4C9BE", background: "var(--surface)",
  fontSize: 15, fontFamily: "var(--font)", outline: "none"
};

function btnStyle(variant) {
  const base = {
    padding: "10px 18px", borderRadius: "var(--radius-sm)",
    fontSize: 14, fontFamily: "var(--font)", cursor: "pointer", border: "none"
  };
  if (variant === "primary") return { ...base, background: "#2C2C2A", color: "#fff" };
  if (variant === "outline") return { ...base, background: "transparent", border: "0.5px solid #2C2C2A", color: "#2C2C2A" };
  return { ...base, background: "transparent", color: "var(--text-muted)" };
}
```

```jsx
function NudgeResult({ result }) {

    safe: { bg: "#D4EDD4", border: "#A8D5A2", text: "#1A4D1A" },

    <div style={{ marginTop: 20, background: c.bg, border: `0.5px solid ${c.border}`,
      borderRadius: "var(--radius-lg)", padding: "20px" }}>
      <div style={{ fontSize: 15, color: c.text, lineHeight: 1.5 }}>
        {result.nudge?.message}
      </div>

              style={{ marginLeft: 10, color: "#2C2C2A", textDecoration: "underline" }}>
              View →
            </a>
          )}
        </div>
      ))}
    </div>
  );
}
```

```js
  res.json({ nudge, alternatives: wireResults.alternatives || [], budgetImpact: pace });
});

export default router;
```

## 8. Anakin.io Wire Orchestrator

```js
// backend/engine/wireOrchestrator.js
import { getCached, setCache } from "../middleware/cache.js";
import { scrapePrice } from "./scraper.js";
import dotenv from "dotenv";
dotenv.config({ path: "../../.env" });

const BASE = process.env.ANAKIN_BASE_URL;   // e.g. https://api.anakin.io
const KEY  = process.env.ANAKIN_API_KEY;
const DEMO = process.env.DEMO_MODE === "true" || !KEY;

// ⚠️  Verify these action IDs in your anakin.io Wire catalogue before running
const CATEGORY_WIRE_MAP = {
  electronics:  ["amazon-product", "flipkart-search"],
  shopping:     ["amazon-product", "flipkart-search"],
  ordering_in:  ["zomato-orders",  "swiggy-orders"],
  groceries:    ["blinkit-search", "bigbasket-search"],
  default:      ["amazon-product", "flipkart-search"],
};

const inflight = new Map();

async function wireCall(actionId, params) {
  const cacheKey = `${actionId}:${JSON.stringify(params)}`;
  const cached = getCached(cacheKey);
  if (cached) return cached;
  if (inflight.has(cacheKey)) return inflight.get(cacheKey);

  const promise = fetch(`${BASE}/v1/wire/run`, {
    method: "POST",
    headers: { "Authorization": `Bearer ${KEY}`, "Content-Type": "application/json" },
    body: JSON.stringify({ action_id: actionId, params })
  })
    .then(r => { if (!r.ok) throw new Error(`${r.status}`); return r.json(); })
    .then(data => { setCache(cacheKey, data); inflight.delete(cacheKey); return data; })
    .catch(err => { inflight.delete(cacheKey); throw err; });

  inflight.set(cacheKey, promise);
  return promise;
}

function parseWireResponse(actionId, data) {
  const out = data.result || data.data || {};
  if (actionId.includes("amazon") || actionId.includes("flipkart") || actionId.includes("blinkit")) {
    return {
      platform: actionId.includes("amazon") ? "Amazon"
               : actionId.includes("flipkart") ? "Flipkart" : "Blinkit",
      price:  parseFloat(out.price || out.current_price || 0),
      title:  out.title || out.name || "Unknown",
      link:   out.url   || out.link || "#",
    };
  }
  if (actionId.includes("zomato") || actionId.includes("swiggy")) {
    return {
      platform: actionId.includes("zomato") ? "Zomato" : "Swiggy",
      spent:  parseFloat(out.total_spent || 0),
      orders: parseInt(out.order_count   || 0),
    };
  }
  return out;
}

export async function orchestrate(intent) {
  if (DEMO) return getDemoResults(intent);

  const actions = CATEGORY_WIRE_MAP[intent.category] || CATEGORY_WIRE_MAP.default;
  const params  = { query: intent.productName, max_price: intent.price * 1.2 };

  const settled = await Promise.allSettled(
    actions.map(id => wireCall(id, params).then(res => ({ id, res })))
  );

  let parsed = [];
  for (const r of settled) {
    if (r.status === "fulfilled") {
      const p = parseWireResponse(r.value.id, r.value.res);
      if (p) parsed.push(p);
    } else {
      // Wire failed — try Universal Scraper fallback
      console.warn(`Wire ${r.reason?.message} — scraper fallback`);
      const scraped = await scrapePrice(intent.productName, r.reason?.actionId);
      if (scraped) parsed.push(scraped);
    }
  }

  if (!parsed.length) return getDemoResults(intent);

  const alternatives = parsed
    .filter(p => p.price && p.price < intent.price)
    .sort((a, b) => a.price - b.price);

  return {
    alternatives,
    originalPrice: intent.price,
    bestSaving: alternatives[0] ? intent.price - alternatives[0].price : 0
  };
}

export async function validateWireActions() {
  const ids = [...new Set(Object.values(CATEGORY_WIRE_MAP).flat())];
  const valid = [], invalid = [];
  await Promise.allSettled(ids.map(async id => {
    try {
      const r = await fetch(`${BASE}/v1/wire/run`, {
        method: "POST",
        headers: { "Authorization": `Bearer ${KEY}`, "Content-Type": "application/json" },
        body: JSON.stringify({ action_id: id, params: { query: "test" } })
      });
      const d = await r.json();
      (d.error?.toLowerCase().includes("not found") ? invalid : valid).push(id);
    } catch { invalid.push(id); }
  }));
  if (invalid.length) console.error("❌ Invalid Wire IDs:", invalid);
  else console.log("✅ All Wire actions valid:", valid);
  return { valid, invalid };
}

function getDemoResults(intent) {
  return {
    alternatives: [
      { platform: "Flipkart", price: Math.round(intent.price * 0.88), title: intent.productName, link: "#" },
      { platform: "Amazon",   price: Math.round(intent.price * 0.92), title: intent.productName, link: "#" },
    ],
    originalPrice: intent.price,
    bestSaving: Math.round(intent.price * 0.12)
  };
}
```

## 9. Universal Scraper Fallback

```js
// backend/engine/scraper.js
const BASE = process.env.ANAKIN_BASE_URL;
const KEY  = process.env.ANAKIN_API_KEY;

const SCRAPE_TARGETS = {
  "amazon-product":  q => `https://www.amazon.in/s?k=${encodeURIComponent(q)}`,
  "flipkart-search": q => `https://www.flipkart.com/search?q=${encodeURIComponent(q)}`,
  "blinkit-search":  q => `https://blinkit.com/s/?q=${encodeURIComponent(q)}`,
  "zomato-orders":   ()=> `https://www.zomato.com`,
};

export async function scrapePrice(query, actionId) {
  const buildUrl = SCRAPE_TARGETS[actionId];
  if (!buildUrl || !BASE || !KEY) return null;

  try {
    const res = await fetch(`${BASE}/v1/scraper/run`, {
      method: "POST",
      headers: { "Authorization": `Bearer ${KEY}`, "Content-Type": "application/json" },
      body: JSON.stringify({
        url: buildUrl(query),
        prompt: `Extract top 2 results as JSON array: [{title, price (number, INR), url}]. Numbers only for price. Return only JSON.`
      })
    });
    const data = await res.json();
    const text = data.result || data.output || "";
    const clean = text.replace(/```json|```/g, "").trim();
    const items = JSON.parse(clean);
    if (!Array.isArray(items) || !items.length) return null;
    return {
      platform: actionId.includes("amazon") ? "Amazon"
               : actionId.includes("flipkart") ? "Flipkart"
               : actionId.includes("blinkit") ? "Blinkit" : "Web",
      price: parseFloat(items[0].price),
      title: items[0].title,
      link:  items[0].url || "#",
    };
  } catch { return null; }
}
```

## 10. Nudge Engine

```js
// backend/engine/nudgeEngine.js
export function generateNudge(pace, wireResults, intent) {
  const { remaining, daysToSalary, isOverpacing } = pace;
  const { price, productName } = intent;
  const postPurchase = remaining - price;

  // Priority 1: cheaper alternative found via Wire/Scraper
  const cheaper = wireResults.alternatives?.[0];
  if (cheaper && cheaper.price < price) {
    const saving = price - cheaper.price;
    const meals  = Math.round(saving / 150);
    return {
      type: "alternative", severity: "alternative",
      message: `${productName} is ₹${cheaper.price.toLocaleString("en-IN")} on ${cheaper.platform}. You save ₹${saving.toLocaleString("en-IN")} — that's ${meals} meal${meals !== 1 ? "s" : ""}.`,
      action: `View on ${cheaper.platform}`,
    };
  }

  // Priority 2: payday is close
  if (daysToSalary <= 5 && postPurchase < 500)
    return {
      type: "wait", severity: "warning",
      message: `Salary in ${daysToSalary} day${daysToSalary !== 1 ? "s" : ""}. This leaves ₹${postPurchase.toLocaleString("en-IN")} for essentials. Worth waiting?`,
      action: "Remind me on payday",
    };

  // Priority 3: overpacing
  if (isOverpacing && postPurchase < 1000)
    return {
      type: "cooloff", severity: "warning",
      message: `You're spending faster than your plan this month. Give it 2 hours — if you still want it, go for it.`,
      action: "Set a reminder",
    };

  // Priority 4: all good
  if (postPurchase > 2000)
    return {
      type: "safe", severity: "safe",
      message: `You'll have ₹${postPurchase.toLocaleString("en-IN")} left after this. You're on track — go for it.`,
      action: "Proceed",
    };

  return {
    type: "info", severity: "info",
    message: `₹${price.toLocaleString("en-IN")} now. At this rate, that's ₹${(price * 12).toLocaleString("en-IN")}/year on ${productName}.`,
    action: "Still want it?",
  };
}
```

## 11. Cache Middleware

```js
// backend/middleware/cache.js
const cache = new Map();
const TTL = 5 * 60 * 1000;   // 5 minutes — Wire data doesn't need to be fresher

export function getCached(key) {
  const entry = cache.get(key);
  if (!entry) return null;
  if (Date.now() - entry.ts > TTL) { cache.delete(key); return null; }
  return entry.data;
}

export function setCache(key, data) {
  cache.set(key, { data, ts: Date.now() });
}
```

## 12. Shared Budget Engine

```js
// shared/budgetEngine.js  — imported by both frontend and backend
export const ALLOCATIONS = {
  rent:         0.25,
  savings:      0.20,
  groceries:    0.10,
  fun_leisure:  0.10,
  health:       0.05,
  ordering_in:  0.05,
  shopping:     0.05,
  investment:   0.20,
};

export function computeBudget(salary) {
  return Object.fromEntries(
    Object.entries(ALLOCATIONS).map(([k, v]) => [k, Math.round(salary * v)])
  );
}

export function spendingPace(catBudget, spent, dayOfMonth, daysInMonth) {
  const safePace   = (dayOfMonth / daysInMonth) * catBudget;
  const remaining  = catBudget - spent;
  const daysLeft   = daysInMonth - dayOfMonth;
  return {
    safePace,
    remaining,
    daysToSalary: daysLeft,
    isOverpacing: spent > safePace,
    dailyBudgetLeft: daysLeft > 0 ? remaining / daysLeft : 0,
  };
}
```
